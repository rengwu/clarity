import assert from 'node:assert/strict';
import { mkdtemp, mkdir, writeFile, access } from 'node:fs/promises';
import { homedir, tmpdir } from 'node:os';
import { join, resolve } from 'node:path';
import { createRequire } from 'node:module';
const cache = join(homedir(), 'Library/Caches/Clarity/build');
const require = createRequire(join(cache, 'loader.cjs'));
const packaged = process.env.CLARITY_X_PACKAGE === '1';
const evidence = resolve('docs/wiki/evidence/2026-10-02T160132Z-x-default-browser', packaged ? 'package' : 'development');
await mkdir(evidence, {recursive:true});
const root = await mkdtemp(join(tmpdir(), 'clarity-x-browser-'));
const application = await require('playwright-core')._electron.launch({
  executablePath: packaged ? join(cache, 'Clarity.app/Contents/MacOS/Electron') : require('electron'),
  args: packaged ? ['--use-mock-keychain'] : [join(cache, 'output/main.mjs'), '--use-mock-keychain'],
  env: {...process.env, CLARITY_TEST_ROOT:root},
});
const observations = {packaged, checks:[]};
try {
  const page = await application.firstWindow();
  page.setDefaultTimeout(10000);
  await page.locator('.sidebar').waitFor();
  await application.evaluate(({shell, ipcMain}) => {
    globalThis.xFixture = {urls:[], logins:[], mode:'ok'};
    shell.openExternal = async url => {
      const fixture = globalThis.xFixture;
      fixture.urls.push(url);
      if (fixture.mode === 'error') throw new Error('Fixture browser launch failed');
      if (fixture.mode === 'hold') await new Promise(resolve => {fixture.release = resolve;});
    };
    const original = ipcMain._invokeHandlers.get('clarity:call');
    ipcMain.removeHandler('clarity:call');
    ipcMain.handle('clarity:call', (event, method, args) => {
      if (method === 'browser.login') {
        globalThis.xFixture.logins.push(args.url);
        return {ok:true, value:{opened:true}};
      }
      return original(event, method, args);
    });
  });
  await page.evaluate(() => window.clarity.call('settings.save', {enabled:['chat','notes','todos','wiki','bookmarks']}));
  await page.reload();
  await page.locator('.sidebar').getByRole('button', {name:'Settings', exact:true}).click();
  await page.getByRole('navigation', {name:'Settings sections'}).getByRole('button', {name:'X bookmarks',exact:true}).click();
  const section = page.locator('#settings-x');
  const open = section.getByRole('button', {name:'Open X',exact:true});
  await open.click();
  await page.waitForFunction(() => !document.querySelector('#settings-x button').disabled);
  assert.deepEqual(await application.evaluate(() => globalThis.xFixture.urls), ['https://x.com/i/bookmarks']);
  assert.deepEqual(await application.evaluate(() => globalThis.xFixture.logins), []);
  await assert.rejects(access(join(root,'.clarity/browser')));
  observations.checks.push('Open X reaches real external.open IPC and shell.openExternal once; no managed login or profile');
  await application.evaluate(() => {globalThis.xFixture.mode = 'error';});
  await open.click();
  await section.getByRole('alert').filter({hasText:'Fixture browser launch failed'}).waitFor();
  await application.evaluate(() => {globalThis.xFixture.mode = 'hold';});
  await open.click();
  assert.equal(await open.isDisabled(), true);
  assert.equal(await section.getByRole('button',{name:'Sign in for import',exact:true}).isDisabled(), true);
  await open.evaluate(button => button.click());
  assert.equal((await application.evaluate(() => globalThis.xFixture.urls)).length, 3);
  await application.evaluate(() => {globalThis.xFixture.release();globalThis.xFixture.mode='ok';});
  await page.waitForFunction(() => !document.querySelector('#settings-x button').disabled);
  assert.equal(await section.getByRole('alert').count(), 0);
  observations.checks.push('Failure stays in X section; retry clears error; pending launch blocks duplicate and fallback submissions');
  await section.getByRole('button', {name:'Sign in for import',exact:true}).click();
  await page.waitForFunction(() => !document.querySelector('#settings-x button').disabled);
  assert.deepEqual(await application.evaluate(() => globalThis.xFixture.logins), ['https://x.com/i/bookmarks']);
  observations.checks.push('Separate import button retains browser.login dispatch (login stubbed; no live X access)');
  for (const [theme,width,height] of [['dark',1200,800],['light',850,600]]) {
    await page.setViewportSize({width,height});
    await page.evaluate(theme => window.clarity.call('settings.save',{theme}),theme);
    await page.reload();
    await page.locator('.sidebar').getByRole('button',{name:'Settings',exact:true}).click();
    await page.getByRole('navigation',{name:'Settings sections'}).getByRole('button',{name:'X bookmarks',exact:true}).click();
    await open.waitFor();
    assert.equal(await section.evaluate(element => element.scrollWidth <= element.clientWidth),true);
    assert.equal(await open.isVisible(),true);
    assert.equal(await section.getByRole('button',{name:'Sign in for import',exact:true}).isVisible(),true);
    await page.screenshot({path:join(evidence,`${theme}-${width}.png`)});
  }
  observations.checks.push('Both actions remain available after reload; dark desktop and light minimum size fit');
  await writeFile(join(evidence,'observations.json'),JSON.stringify(observations,null,2));
  console.log(JSON.stringify(observations,null,2));
} finally {
  await application.close();
}
