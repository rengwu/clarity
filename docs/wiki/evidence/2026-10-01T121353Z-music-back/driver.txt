import assert from 'node:assert/strict';
import { createRequire } from 'node:module';
import { readFile, writeFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
const require = createRequire('/Users/rengwu/Library/Caches/Clarity/build/loader.cjs');
const evidence = '/Users/rengwu/Library/Caches/Clarity/build/music-back-2026-10-01';
const source = '/Users/rengwu/Desktop/Projects/clarity-spikes/2026-10-01-cursor';
const hash = content => createHash('sha256').update(content).digest('hex');
const files = [];
for (const name of ['app.js','music.js','modules.js','data.js','fixtures.js','ui.js','motion.js','style.css','design.css','index.html','server.mjs']) {
  const content = await readFile(`${source}/${name}`);
  files.push({name, sha256: hash(content)});
  if (name !== 'server.mjs') {
    const response = await fetch(`http://127.0.0.1:4318/${name}`);
    assert.equal(response.status, 200);
    assert.equal(hash(Buffer.from(await response.arrayBuffer())), hash(content));
  }
}
await writeFile(`${evidence}/build.json`, JSON.stringify({recordedAt: new Date().toISOString(), source, files},null,2));
const browser = await require('playwright-core').chromium.launch({
  executablePath: '/Applications/Google Chrome.app/Contents/MacOS/Google Chrome', headless: true,
});
const results = [];
const errors = [];
try {
  const page = await browser.newPage({viewport: {width:1440,height:900},colorScheme:'dark'});
  page.on('pageerror', error => errors.push(error.message));
  await page.goto('http://127.0.0.1:4318/');
  await page.locator('[data-action="navigate"][data-id="music"]').click();
  for (const width of [1440,390]) {
    await page.setViewportSize({width,height:900});
    for (const [tab, itemSelector, label] of [
      ['albums','.album-card','Albums'],['playlists','.album-card','Playlists'],['artists','.artist-card','Artist'],
    ]) {
      await page.locator(`.music-tabs [data-action="x-music-tab"][data-id="${tab}"]`).click();
      await page.locator(itemSelector).first().click();
      const control = page.getByRole('button',{name:`Back to ${label}`,exact:true});
      const text = control.locator('span');
      const box = await control.boundingBox();
      assert.ok(box.width > 50 && box.height >= 24);
      await text.hover();
      if (tab === 'albums' && width === 1440) await page.screenshot({path:`${evidence}/albums-back.png`});
      await text.click();
      assert.equal(await page.locator('.music-back').count(),0);
      assert.equal(await page.locator(`.music-tabs [data-action="x-music-tab"][data-id="${tab}"]`).getAttribute('aria-pressed'),'true');
      results.push({width,tab,activation:'label click',passed:true,control:box});
    }
  }
  await page.setViewportSize({width:1440,height:900});
  await page.locator('.music-tabs [data-action="x-music-tab"][data-id="artists"]').click();
  await page.locator('.artist-card').first().click();
  const artist = await page.locator('.artist-heading h1').textContent();
  await page.locator('.album-card').first().click();
  await page.locator('.music-back span').click();
  assert.equal(await page.locator('.artist-heading h1').textContent(),artist);
  const back = page.locator('.music-back button');
  await back.focus();
  await page.keyboard.press('Tab');
  await page.keyboard.press('Shift+Tab');
  assert.equal(await back.evaluate(element => element === document.activeElement),true);
  assert.notEqual(await back.evaluate(element => getComputedStyle(element).outlineStyle),'none');
  await page.keyboard.press('Enter');
  assert.equal(await page.locator('.music-back').count(),0);
  results.push({scenario:'Nested album → artist → artists; label and keyboard Enter',passed:true});
  await page.locator('.music-tabs [data-action="x-music-tab"][data-id="albums"]').click();
  await page.locator('.album-card').first().click();
  await page.locator('.music-back button').focus();
  await page.keyboard.press('Space');
  assert.equal(await page.locator('.music-back').count(),0);
  await page.locator('.album-card').first().click();
  await page.locator('.music-back svg').click();
  assert.equal(await page.locator('.music-back').count(),0);
  results.push({scenario:'Keyboard Space and chevron click',passed:true});
  const player = await page.evaluate(() => JSON.parse(localStorage.getItem('clarity-cursor-spike-v1')).player);
  assert.equal(player.playing,false);
  assert.deepEqual(errors,[]);
  console.log('Passed: label clicks on Albums, Playlists and Artist at desktop/mobile; nested Back; Enter/Space; arrow click; no playback or runtime errors.');
} finally {
  await writeFile(`${evidence}/observations.json`,JSON.stringify({results,errors},null,2));
  await browser.close();
}
